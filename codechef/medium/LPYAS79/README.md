```py
# Given dictionary
student_grades = {"Alice": 85, "Bob": 72, "Charlie": 90, "David": 65, "Eva": 88, "John": 45}

# Complete the code 
t=input()
if t in student_grades:
    print(student_grades[t])
else:
    print("Not Found")
```
